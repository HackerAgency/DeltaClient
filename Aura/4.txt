private void onUpdate(UpdateEvent event) {

    float[] ranges = getRanges();

        this.resetAllVectors();
        return;
    }
    
    // 4. Resolve всех игроков (для античита)
    for (Entity entity : mc.world.getAllEntities()) {
        if (entity instanceof AbstractClientPlayerEntity abstractClientPlayerEntity) {
            abstractClientPlayerEntity.resolve();
        }
    }
    
    // 5. Расчет и применение ротации
    if (elytraTarget.canTarget(target)) {
        update(target); // Elytra режим
    } else {
        if (rotationModeImpl.is("Без ротации")) {
            pro_rotation.rotatePrimitiveToSend(null);
            this.resetAllVectors();
        } else if (rotationModeImpl.is("ФанТайм")) {
            funTimeRotation(); // Фан режим
        } else {
            // Обычные режимы ротации
            final URotate.Vector3d2fRB rotationData = pro_rotation.updateRotationData(
                rotationModeImpl.get(), 
                lastHandledVectorToRotation == null ? 
                    new Vector2f(mc.player.rotationYaw, mc.player.rotationPitch) : 
                    lastHandledVectorToRotation, 
                ranges[2]
            );
            if (rotationData.isToUse()) {
                this.updateAllVectors(rotationData.getVec3d(), rotationData.getVec2f());
                pro_rotation.rotatePrimitiveToSend(lastHandledVectorToRotation);
            } else {
                this.resetAllVectors();
            }
        }
    }
    
    // 6. Anti-miss система
    UAura.antiMissesHittingUpdate(target, cpsBypass.get(), this.isRayCastRuleToAttack(), postMissSpamming.get());
    
    // 7. Проверка возможности атаки
    boolean canAttack = UAura.shouldAttack(target, this.isRayCastRuleToAttack(), 
        !rotationModeImpl.is("ФанТайм"), cooldownMode.is("1.9 или выше"), 0L, ranges);
    
    // 8. Дополнительная проверка для ФанТайм
    if (rotationModeImpl.is("ФанТайм")) {
        canAttack = canAttack && anyEntityOnRay(target, lerpRotation.x, lerpRotation.y, ranges[2]);
    }

        // Release всех игроков если нельзя атаковать
        for (Entity entity : mc.world.getAllEntities()) {
            if (entity instanceof AbstractClientPlayerEntity abstractClientPlayerEntity) {
                abstractClientPlayerEntity.releaseResolver();
            }
        }
        return;
    }
    
    // 9. Проверка поедания
    if (mc.player.isEating()) {
        if (eatFixMode.is("Прекратить бить")) return;
        else if (eatFixMode.is("NCP обход"))
            new CPlayerDiggingPacket(CPlayerDiggingPacket.Action.RELEASE_USE_ITEM, 
                new BlockPos(-1, -1, -1), Direction.DOWN).sendSilent();
    }

    final Runnable[]
        shieldBreak = UAura.hitShieldBreakTaskForUse(target, shieldBreaker.get()),
        shieldPressBypass = UAura.resetShieldSilentTaskForUse(true),
        skipSilentSprint = UAura.skipSilentSprintingTaskForUse(!stopSprint.is("Никогда"));
    
    final Runnable preHitSendCodeSingleTick = () -> {
        skipSilentSprint[0].run();
        shieldPressBypass[0].run();
        shieldBreak[0].run();
    };
    
    final Runnable postHitSendCodeSingleTick = () -> {
        shieldBreak[1].run();
        shieldPressBypass[1].run();
        skipSilentSprint[1].run();
    };

    UAura.useEntity(target, preHitSendCodeSingleTick, postHitSendCodeSingleTick, Hand.MAIN_HAND, cpsBypass.get());
    
    // 12. Release всех игроков после атаки
    for (Entity entity : mc.world.getAllEntities()) {
        if (entity instanceof AbstractClientPlayerEntity abstractClientPlayerEntity) {
            abstractClientPlayerEntity.releaseResolver();
        }
    }
}

public float[] getRanges() {

        float groundDistance = this.target != null ? 
            (this.target.isElytraFlying() ? 6 : attackRange.get()) : attackRange.get();
        return new float[]{groundDistance, 64F};
    }
    return switch (rangeModeImpl.get()) {
        case "Сильный античит" -> new float[]{3.F, .4F};
        case "NCP или аналоги" -> new float[]{4.05F, 2.7F};
        case "Античит AAC" -> new float[]{3.56F, 1.15F};
        case "Античит Matrix" -> new float[]{3.3F, .65F};
        case "Слабый античит" -> new float[]{6.F, .0F};
        case "Свои настройки" -> new float[]{attackRange.get(), preRange.get()};
        default -> new float[]{3.F, .1F};
    };
}

public static long getMsCooldown() {

        // 1.9+ combat system - основано на атрибуте ATTACK_SPEED
        double attributeAttackSpeed = mc.player.getAttributeValue(Attributes.ATTACK_SPEED);
        float maxDeviation = .2F;
        msCooldown = (long) ((1.F / attributeAttackSpeed) * 1000.F * (1.F - Math.min(maxDeviation, 1.F)));
        
        // Special case для определенных значений скорости
        if (attributeAttackSpeed == 4.D || attributeAttackSpeed == 4.4000000059604645 || 
            attributeAttackSpeed == 4.800000011920929)
            msCooldown = 450L;

        msCooldown = 45L; // 1.8 combat system

    // CPS bypass - добавляет 50ms к каждому 7-му удару
    if (UAura.cpsBypassTrigger()) msCooldown += 50L;
    
    return msCooldown;

public static boolean shouldAttack(LivingEntity livingTarget, boolean rayCast, boolean distanceCheck, boolean fallCheck, long cooldownMSOffset, float[] ranges) {

    if (distanceCheck && livingTarget != null && !mc.player.validDistance(livingTarget, ranges[0], true))
        return false;
    
    // Проверка кулдауна
    if (!UAura.msCooldownReached(cooldownMSOffset)) return false;
    
    // Проверка момента для крита
    boolean validNext = UAura.isBestMomentToHit(fallCheck);
    
    // Raycast проверка
    if (validNext && rayCast && !anyEntityOnRay(livingTarget, ranges[0]))
        validNext = false;
    
    return validNext;
}

public static void antiMissesHittingUpdate(LivingEntity targetIn, boolean cpsBypass, boolean rayCastCheck, boolean enabled) {
    if (targetIn == null || counterTo0PostMissHits == 0 || !enabled || targetIn.hurtTime != 0)
        antiMissesHittingReset();
    
    if (enabled && targetIn != null && UAura.msCooldownHasMs(cpsBypassTrigger() ? 250 : 150) && mc.player.isSwingInProgress) {
        // Обнаружение промаха: игрок свингает, но цель не получила урон
        if (!missDetected && counterTo0PostMissHits == 0 && targetIn.hurtTime == 0) {
            missDetected = true;
            counterTo0PostMissHits = maxHitsCountOnMiss();
        }
        
        // Закликивание при промахе
        if (missDetected && counterTo0PostMissHits > 0 && targetIn != null) {
            if ((!rayCastCheck || anyEntityOnRay(targetIn, 6.F)) && 
                useEntity(targetIn, () -> {}, () -> {}, Hand.MAIN_HAND, cpsBypass)) 
                --counterTo0PostMissHits;
        }
    }
}

public static void updateTargetList() {
    if (mc.world == null) {
        synchronized (entityList) {
            entityList.clear();
        }
        return;
    }
 
    List<Entity> entities = mc.world.loadedEntityList()
            .stream()

        entityList.addAll(entities);

private static final Predicate<Entity> ENTITY_FILTER = TargetComponent::isValid;
 
private static boolean isValid(final Entity entity) {
    Targets module = Client.inst().featureManager().targets;
    if (entity == null) return false;
    if (entity instanceof FakePlayer) return false;
    if (!module.throughWalls().get() && !mc.player.canEntityBeSeen(entity)) return false;
    
    if (entity instanceof LivingEntity living) {
        if (living.getHealth() <= 0 || !entity.isAlive() || entity.equals(mc.player)) return false;
    }
    
    if (entity instanceof ArmorStandEntity) return false;
    
    final MultiBooleanSetting setting = module.targets();
    if (!setting.enabled("Невидимые") && entity.isInvisible()) return false;
    
    if (entity instanceof PlayerEntity player) {
        if (Client.inst().featureManager().antiBot.isBot(player)) return false;
        if (Client.inst().friendManager().isFriend(player.getGameProfile().getName())) return false;
        if (!setting.enabled("Игроки")) return false;
        if (!setting.enabled("Голые") && player.getTotalArmorValue() <= 0) return false;
        if (!setting.enabled("Тиммейты") && mc.player.isOnSameTeam(entity)) return false;
    }
    
    if (entity instanceof AnimalEntity) return setting.enabled("Животные");
    if (entity instanceof MobEntity) return setting.enabled("Мобы");
    
    return true;
}

public static List<LivingEntity> getTargets(final double range, boolean saveRange) {
    if (countLoadedEntities != mc.world.getCountLoadedEntities()) {
        executorService.execute(TargetComponent::updateTargetList);
        countLoadedEntities = mc.world.getCountLoadedEntities();
    }
 
    if (currentTarget != null && !isValid(currentTarget)) {
        currentTarget = null;
    }
    if (saveRange) lastRange = range;
    
    synchronized (entityList) {
        List<LivingEntity> list = new ArrayList<>();
        for (Entity entity : entityList) {
            if (entity instanceof LivingEntity living && ENTITY_FILTER.test(entity)) {
                if (mc.player.validDistance(entity, (float) range, true) && 
                    mc.world.loadedLivingEntityList().contains(entity)) {
                    list.add(living);
                }
            }
        }
        return list;
    }
}

private static Optional<LivingEntity> find(final double range, boolean saveRange) {
    List<LivingEntity> validTargets = getTargets(range, saveRange);
    final ModeSetting sortMode = Client.inst().featureManager().targets.sortMode();
    
    if (validTargets.isEmpty()) return Optional.empty();
    
    if (sortMode.is("Адаптивный")) {
        // Приоритет: броня -> FOV -> дистанция -> здоровье
        validTargets.sort(Comparator.comparingDouble(TargetComponent::compareArmor)
                .thenComparingDouble(AuraUtil::calculateFOVFromCamera)
                .thenComparingDouble(AuraUtil::getStrictDistance)
                .thenComparingDouble(PlayerUtil::getEntityHealth));
    } else if (sortMode.is("Дистанция")) {
        validTargets.sort(Comparator.comparingDouble(mc.player::getDistance)
                .thenComparingDouble(PlayerUtil::getEntityHealth));
    } else if (sortMode.is("Здоровье")) {
        validTargets.sort(Comparator.comparingDouble(PlayerUtil::getEntityHealth)
                .thenComparingDouble(AuraUtil::getStrictDistance));
    } else if (sortMode.is("Наводка")) {
        validTargets.sort(Comparator.comparingDouble(AuraUtil::calculateFOVFromCamera)
                .thenComparingDouble(AuraUtil::getStrictDistance)
                .thenComparingDouble(PlayerUtil::getEntityHealth)
                .thenComparingDouble(TargetComponent::compareArmor));
    }
    
    return Optional.of(validTargets.get(0));
}

public Vector3f calculateViewVector(float yaw, float pitch) {
    float pitchRad = (float) (pitch * (Math.PI / 180F));
    float yawRad = (float) (-yaw * (Math.PI / 180F));
    float cosYaw = MathHelper.cos(yawRad);
    float sinYaw = MathHelper.sin(yawRad);
    float cosPitch = MathHelper.cos(pitchRad);
    float sinPitch = MathHelper.sin(pitchRad);
 
    return new Vector3f(sinYaw * cosPitch, -sinPitch, cosYaw * cosPitch);
}
 
public Vector3d getVectorForRotation(float pitch, float yaw) {
    float yawRadians = -yaw * ((float) Math.PI / 180) - (float) Math.PI;
    float pitchRadians = -pitch * ((float) Math.PI / 180);
 
    float cosYaw = MathHelper.cos(yawRadians);
    float sinYaw = MathHelper.sin(yawRadians);
    float cosPitch = -MathHelper.cos(pitchRadians);
    float sinPitch = MathHelper.sin(pitchRadians);
 
    return new Vector3d(sinYaw * cosPitch, sinPitch, cosYaw * cosPitch);
}

private static boolean updateRotation(Rotation rotation, float turnSpeed) {
    Rotation currentRotation = new Rotation(mc.player);
 
    float yawDelta = wrapDegrees(rotation.getYaw() - currentRotation.getYaw());
    float pitchDelta = rotation.getPitch() - currentRotation.getPitch();
 
    float totalDelta = Math.abs(yawDelta) + Math.abs(pitchDelta);
 
    // Распределение скорости между осями пропорционально дельте
    float yawSpeed = (totalDelta == 0) ? 0 : Math.abs(yawDelta / totalDelta) * turnSpeed;
    float pitchSpeed = (totalDelta == 0) ? 0 : Math.abs(pitchDelta / totalDelta) * turnSpeed;
 
    // Применение с GCD патчем
    mc.player.rotationYaw += SensUtil.getSens(clamp(yawDelta, -yawSpeed, yawSpeed));
    mc.player.rotationPitch = MathHelper.clamp(
        mc.player.rotationPitch + SensUtil.getSens(clamp(pitchDelta, -pitchSpeed, pitchSpeed)), 
        -90F, 90F
    );
 
    Rotation finalRotation = new Rotation(mc.player);
    idleTicks = 0;
 
    // Возвращает true если достигли цели
    return finalRotation.getDelta(rotation) < 
           (currentTask.equals(RotationTask.RESET) ? currentReturnSpeed : currentTurnSpeed);
}

private Vector3d2fRB updateRotationData(String rotationName, RotateRules rules,

    boolean behindWall, Vector3d rotateTo, Vector2f rotation, boolean prevRayCasted) {

        return new Vector3d2fRB(rotateTo, rotation, false);

    // Проверка условий работы ротации
    if (rules.isRotOnlyPreHitTick() && !(preHitTick || UAura.msCooldownReached()) || 
        rules.isRotOnlyHitTick() && !(hitTick || UAura.msCooldownReached(50))) {
        return new Vector3d2fRB(rotateTo, rotation, false);
    }
 
    // Копирование ротации для изоляции
    rotation = new Vector2f(rotation.x, rotation.y);

    float easingPC = rules.getEaseStrengthPercent();
    boolean easingIsOpposite = rules.isEasingOpposite();
    double posRandomValue = rules.getVectorPointPosRandom0rec();

                           (rules.isBehindWallTurnAway() && behindWall && preHitTick);

                              rules.isPauseRotationOnHitTick() && hitTick ||

    if (posRandomValue > 0.F)
        rotateTo = rotateTo.add(randomAroundD2(posRandomValue), randomAroundD2(posRandomValue), randomAroundD2(posRandomValue));
 
    // Расчет базовых углов
    Vector2f virtRotateVanilla = getVanillaRotate(rotateTo);
    
    // Отводка перед ударом
    if (forceTurnAway) 
        virtRotateVanilla = getTurnAwayOfRotate(virtRotateVanilla, entityTarget, 6.F, 5);
    
    // Добавление рандомизации
    float addRandomYaw = rules.getRandRotDragYaw045() == 0.F ? 0.F : randomAroundF2(rules.getRandRotDragYaw045());
    float addRandomPitch = rules.getRandRotDragPitch030() == 0.F ? 0.F : randomAroundF2(rules.getRandRotDragPitch030());
    
    if (addRandomYaw != 0.F || addRandomPitch != 0.F) {
        virtRotateVanilla.x += addRandomYaw;
        virtRotateVanilla.y += addRandomPitch;
    }
    
    // Расчет скорости
    float yawDiffAtPrev = RotationUtil.getAngleDifference(rotation.x, virtRotateVanilla.x);
    float pitchDiffAtPrev = Math.abs(rotation.y - virtRotateVanilla.y);
    
    float yawDiffPC01 = Math.min(yawDiffAtPrev / 180.F, 1.F);
    float pitchDiffPC01 = Math.min(pitchDiffAtPrev / 90.F, 1.F);
    
    // Применение изинга
    float yawSpeedPC01Calc = yawDiffPC01;
    float pitchSpeedPC01Calc = pitchDiffPC01;
    
    if (easing != null && easingPC > 0.F) {
        if (easingIsOpposite) {
            yawSpeedPC01Calc = lerp(1.F - yawDiffPC01, (float) easing.ease(1.F - yawDiffPC01), easingPC);
            pitchSpeedPC01Calc = lerp(1.F - pitchDiffPC01, (float) easing.ease(1.F - pitchDiffPC01), easingPC);
        } else {
            yawSpeedPC01Calc = lerp(yawDiffPC01, (float) easing.ease(yawDiffPC01), easingPC);
            pitchSpeedPC01Calc = lerp(pitchDiffPC01, (float) easing.ease(pitchDiffPC01), easingPC);
        }
    }
    
    // Взаимоисключение скорости осей
    float mixYawSpeedAndPitchSpeedPC01 = Math.min(rules.getMutualExclusionXYSpeedMul01(), 1.F);
    if (mixYawSpeedAndPitchSpeedPC01 > 0.F) {
        float yawSpeedPC01CalcCopy = yawSpeedPC01Calc;
        yawSpeedPC01Calc *= 1.F - pitchSpeedPC01Calc * mixYawSpeedAndPitchSpeedPC01;
        pitchSpeedPC01Calc *= 1.F - yawSpeedPC01CalcCopy * mixYawSpeedAndPitchSpeedPC01;
    }
    
    // Финальная скорость
    float minYawSpeed = rules.getMinYawSpeed0180();
    float maxYawSpeed = rules.getMaxYawSpeed0180();
    float minPitchSpeed = rules.getMinPitchSpeed090();
    float maxPitchSpeed = rules.getMaxPitchSpeed090();
    
    float yawSpeed = lerp(minYawSpeed, maxYawSpeed, yawSpeedPC01Calc) * speedMul;
    float pitchSpeed = lerp(minPitchSpeed, maxPitchSpeed, pitchSpeedPC01Calc) * speedMul;
    
    // Применение скорости
    final float yawDelta = wrapDegrees(virtRotateVanilla.x - rotation.x);
    final float pitchDelta = virtRotateVanilla.y - mc.player.rotationPitch;
    
    if (!forceStopRotation) {
        rotation.x = wrapDegrees(rotation.x + clamp(yawDelta, -yawSpeed, yawSpeed));
        rotation.y = mc.player.rotationPitch + clamp(pitchDelta, -pitchSpeed, pitchSpeed);
    }
    
    // GCD патч
    if (gcdFix) {
        rotation.x = SensUtil.getSens(rotation.x);
        rotation.y = SensUtil.getSens(rotation.y);
    }
    
    return new Vector3d2fRB(rotateTo, rotation, true);
}
